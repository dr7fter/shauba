### 3.2 TodayView Hooks 拆分

#### 3.2.1 useKeyboardShortcuts Hook

**目标**：将 TodayView 内的键盘事件处理逻辑（行 842-876）提取为独立 hook。

**文件位置**：`src/hooks/useKeyboardShortcuts.ts`

**接口设计**：

```typescript
// src/hooks/useKeyboardShortcuts.ts
import { useEffect } from 'react'

export interface KeyboardShortcutActions {
  reveal: () => void
  submitIfReady: () => void
  chooseRating: (rating: number) => void
  toggleOption: (option: string) => void
  skip: () => void
  startVariant: () => void
  previousQuestion: () => void
  nextQuestion: () => void
  revealed: boolean
}

export function useKeyboardShortcuts(actions: KeyboardShortcutActions) {
  useEffect(() => {
    const onKey = (event: KeyboardEvent) => {
      if (event.ctrlKey || event.metaKey || event.altKey) return
      const tag = (event.target as HTMLElement | null)?.tagName
      if (tag === 'BUTTON' || tag === 'INPUT' || tag === 'TEXTAREA' || tag === 'SELECT')
        return

      const keyUpper = event.key.toUpperCase()
      if (event.key === 'Enter' || event.key === ' ') {
        event.preventDefault()
        if (!actions.revealed) actions.reveal()
        else actions.submitIfReady()
      } else if (event.key >= '1' && event.key <= '4') {
        event.preventDefault()
        actions.chooseRating(Number(event.key))
      } else if (['A', 'B', 'C', 'D'].includes(keyUpper)) {
        event.preventDefault()
        actions.toggleOption(keyUpper)
      } else if (keyUpper === 'S') {
        event.preventDefault()
        actions.skip()
      } else if (keyUpper === 'V' && actions.revealed) {
        event.preventDefault()
        actions.startVariant()
      } else if (event.key === 'ArrowLeft') {
        event.preventDefault()
        actions.previousQuestion()
      } else if (event.key === 'ArrowRight') {
        event.preventDefault()
        actions.nextQuestion()
      }
    }
    window.addEventListener('keydown', onKey)
    return () => window.removeEventListener('keydown', onKey)
  }, [actions])
}
```

**在 TodayView 中使用**：

```typescript
// src/views/TodayView.tsx
import { useKeyboardShortcuts } from '../hooks/useKeyboardShortcuts'

// ... 组件内部
useKeyboardShortcuts({
  reveal,
  submitIfReady,
  chooseRating,
  toggleOption: (opt) => {
    if (!current || current.questionType === 'subjective') return
    setSelected((prev) =>
      current.questionType === 'multiple_choice'
        ? prev.includes(opt)
          ? prev.filter((x) => x !== opt)
          : [...prev, opt].sort()
        : [opt]
    )
  },
  skip,
  startVariant: () => {
    if (current) onStartVariant(current.id)
  },
  previousQuestion,
  nextQuestion,
  revealed,
})
```

**预期收益**：
- TodayView.tsx 减少约 35 行代码
- 键盘快捷键逻辑可复用到其他视图
- 单元测试更容易编写
